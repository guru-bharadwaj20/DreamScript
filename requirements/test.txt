# What the suite needs and the product does not.
#
# These four lived in `base.txt`, and that was a deliberate fix for a real bug: CI installs the
# runtime layers, not `dev.txt`, so a test dependency parked in `dev.txt` is a test that never
# runs on a runner. `piexif` was exactly that, and moving it to `base.txt` made it run.
#
# It also put pytest, hypothesis, psutil and piexif into the **serving image**, because the
# Dockerfile installs `base.txt`. A container that answers /predict has no use for a property-
# based testing library, and shipping one is surface a deployment did not ask for.
#
# So: a third layer, installed by CI and by `make env` and not by the Dockerfile. Same property
# as before - a test dependency is in a layer the test job installs - without the consequence.
pytest==8.3.4
hypothesis==6.168.0    # tests/test_pipeline_fuzz.py
psutil==7.2.2          # tests/test_rl_sandbox.py
piexif==1.1.3          # tests/test_preprocess_exif.py
